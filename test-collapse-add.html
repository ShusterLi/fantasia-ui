使用示例代码

<template>
<f-collapse v-model="activeNames"
:showAddButton="true"
addButtonPosition="top"
addButtonText="添加新项目"
@add="handleAdd">
<f-collapse-item name="1" title="项目一">
这是第一个项目的内容
</f-collapse-item>
<f-collapse-item name="2" title="项目二">
这是第二个项目的内容
</f-collapse-item>
</f-collapse>
</template>

<script setup lang="ts">
import { ref } from 'vue'

const activeNames = ref([])
const items = ref([
{ name: '1', title: '项目一', content: '这是第一个项目的内容' },
{ name: '2', title: '项目二', content: '这是第二个项目的内容' }
])

function handleAdd() {
const newName = String(items.value.length + 1)
items.value.push({
name: newName,
title: `项目${newName}`,
content: `这是第${newName}个项目的内容`
})
}
</script>

API 文档

// 新增的 props
interface FCollapseProps {
accordion?: boolean
showAddButton?: boolean // 是否显示新增按钮
addButtonPosition?: 'top' | 'bottom' // 按钮位置，默认为 'bottom'
addButtonText?: string // 按钮文本，默认为 '新增选项'
}

// 新增的事件
@add - 点击新增按钮时触发
// 示例：@add="handleAdd"

// 默认值
const props = withDefaults(defineProps<FCollapseProps>(), {
accordion: false,
showAddButton: false,
addButtonPosition: 'bottom',
addButtonText: '新增选项'
})
